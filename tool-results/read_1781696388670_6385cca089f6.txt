     1→'use client';
     2→

     4→import { useAppStore } from '@/store/app-store';
     5→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
     6→import { Button } from '@/components/ui/button';
     7→import { Input } from '@/components/ui/input';
     8→import { Textarea } from '@/components/ui/textarea';
     9→import { Label } from '@/components/ui/label';
    10→import { Checkbox } from '@/components/ui/checkbox';
    11→import { Badge } from '@/components/ui/badge';
    12→import { Skeleton } from '@/components/ui/skeleton';
    13→import { Separator } from '@/components/ui/separator';
    14→import {
    15→  Select,
    16→  SelectContent,
    17→  SelectItem,
    18→  SelectTrigger,
    19→  SelectValue,
    20→} from '@/components/ui/select';
    21→import {
    22→  AlertDialog,
    23→  AlertDialogAction,
    24→  AlertDialogCancel,
    25→  AlertDialogContent,
    26→  AlertDialogDescription,
    27→  AlertDialogFooter,
    28→  AlertDialogHeader,
    29→  AlertDialogTitle,
    30→} from '@/components/ui/alert-dialog';
    31→import {
    32→  Dialog,
    33→  DialogContent,
    34→  DialogDescription,
    35→  DialogFooter,
    36→  DialogHeader,
    37→  DialogTitle,
    38→} from '@/components/ui/dialog';
    39→import { RadioGroup, RadioGroupItem } from '@/components/ui/radio-group';
    40→import { ArrowLeft, Save, Send, RotateCcw, Loader2, AlertTriangle } from 'lucide-react';
    41→import { toast } from 'sonner';
    42→import type {
    43→  AppraisalFormDataFull,
    44→  AssignmentDetail,
    45→  AchievementGoalItem,
    46→  CompetencyItem,
    47→  ExplanationItem,
    48→  FutureGoalItem,
    49→  RemarksData,
    50→  AppraisalStatus,
    51→} from '@/lib/types';
    52→import {
    53→  GOAL_RATING_OPTIONS,
    54→  COMPETENCY_RATING_OPTIONS,
    55→  TECHNICAL_SKILLS,
    56→  LEADERSHIP_SKILLS,
    57→  MANAGERIAL_SKILLS,
    58→  SECTION_DESCRIPTIONS,
    59→  createDefaultFormData,
    60→  calculateAppraisalScores,
    61→} from '@/lib/constants';
    62→import { APPRAISAL_STATUS_LABELS, APPRAISAL_STATUS_COLORS } from '@/lib/constants';
    63→
    64→export default function AppraisalForm() {
    65→  const { viewParams, currentUser, setCurrentView } = useAppStore();
    66→  const assignmentId = viewParams?.id;
    67→
    68→  const [assignment, setAssignment] = useState<AssignmentDetail | null>(null);
    69→  const [formData, setFormData] = useState<AppraisalFormDataFull | null>(null);
    70→  const [loading, setLoading] = useState(true);
    71→  const [saving, setSaving] = useState(false);
    72→  const [submitting, setSubmitting] = useState(false);
    73→  const [returnDialog, setReturnDialog] = useState(false);
    74→  const [returnReason, setReturnReason] = useState('');
    75→  const [returning, setReturning] = useState(false);
    76→  const [error, setError] = useState<string | null>(null);
    77→
    78→  const userRole = currentUser?.role || 'employee';
    79→  const currentActionBy = assignment?.currentActionBy;
    80→
    81→  // Can the user edit?
    82→  const canEditEmployee = userRole === 'employee' && currentActionBy === 'employee';
    83→  const canEditSupervisor = userRole === 'supervisor' && (currentActionBy === 'supervisor' || currentActionBy === 'employee');
    84→  const canEditHR = (userRole === 'admin' || userRole === 'hr') && currentActionBy === 'hr';
    85→  const canEditManagement = userRole === 'management' && (currentActionBy === 'management');
    86→  const canSubmit = currentActionBy === userRole;
    87→
    88→  // Fetch assignment and form data
    89→  useEffect(() => {
    90→    if (!assignmentId) return;
    91→    async function fetchData() {
    92→      try {
    93→        setError(null);
    94→        const [assignRes, formRes] = await Promise.all([
    95→          fetch(`/api/assignments/${assignmentId}`),
    96→          fetch(`/api/assignments/${assignmentId}/form`),
    97→        ]);
    98→        if (assignRes.ok) setAssignment(await assignRes.json());
    99→        if (formRes.ok) {
   100→          const data = await formRes.json();
   101→          setFormData(data.formData || createDefaultFormData());
   102→        } else {
   103→          setFormData(createDefaultFormData());
   104→        }
   105→      } catch {
   106→        console.warn('Server unavailable, could not load appraisal form');
   107→        setError('Unable to load appraisal form. The server may be temporarily unavailable.');
   108→      } finally {
   109→        setLoading(false);
   110→      }
   111→    }
   112→    fetchData();
   113→  }, [assignmentId]);
   114→
   115→  // Update form helpers
   116→  const updateAchievement = useCallback(
   117→    (section: 'achievements' | 'goals', index: number, field: keyof AchievementGoalItem, value: string | number) => {
   118→      setFormData((prev) => {
   119→        if (!prev) return prev;
   120→        const updated = [...prev[section]];
   121→        updated[index] = { ...updated[index], [field]: value };
   122→        return { ...prev, [section]: updated };
   123→      });
   124→    },
   125→    []
   126→  );
   127→
   128→  const updateSkill = useCallback(
   129→    (section: 'technicalSkills' | 'leadershipSkills' | 'managerialSkills', index: number, field: 'employeeRating' | 'supervisorRating', value: number | 'NA') => {
   130→      setFormData((prev) => {
   131→        if (!prev) return prev;
   132→        const updated = [...prev[section]];
   133→        updated[index] = { ...updated[index], [field]: value };
   134→        return { ...prev, [section]: updated };
   135→      });
   136→    },
   137→    []
   138→  );
   139→
   140→  const updateExplanation = useCallback(
   141→    (index: number, field: keyof ExplanationItem, value: string | boolean | number) => {
   142→      setFormData((prev) => {
   143→        if (!prev) return prev;
   144→        const updated = [...prev.explanations];
   145→        updated[index] = { ...updated[index], [field]: value };
   146→        return { ...prev, explanations: updated };
   147→      });
   148→    },
   149→    []
   150→  );
   151→
   152→  const updateGoal = useCallback(
   153→    (index: number, field: keyof FutureGoalItem, value: string | boolean | [boolean, boolean, boolean, boolean], quarterIdx?: number) => {
   154→      setFormData((prev) => {
   155→        if (!prev) return prev;
   156→        const updated = [...prev.futureGoals];
   157→        if (field === 'quarters' && quarterIdx !== undefined) {
   158→          const newQuarters = [...updated[index].quarters] as [boolean, boolean, boolean, boolean];
   159→          newQuarters[quarterIdx] = value as boolean;
   160→          updated[index] = { ...updated[index], quarters: newQuarters };
   161→        } else {
   162→          updated[index] = { ...updated[index], [field]: value };
   163→        }
   164→        return { ...prev, futureGoals: updated };
   165→      });
   166→    },
   167→    []
   168→  );
   169→
   170→  const updateRemarks = useCallback(
   171→    (section: 'hr' | 'supervisor', field: string, value: boolean | string | null) => {
   172→      setFormData((prev) => {
   173→        if (!prev) return prev;
   174→        const updated = { ...prev.remarks };
   175→        if (section === 'hr') {
   176→          (updated as any)[`hr${field}`] = value;
   177→        } else {
   178→          (updated as any)[`supervisor${field}`] = value;
   179→        }
   180→        return { ...prev, remarks: updated as RemarksData };
   181→      });
   182→    },
   183→    []
   184→  );
   185→
   186→  // Calculate scores
   187→  const employeeScores = useMemo(
   188→    () => (formData ? calculateAppraisalScores(formData, 'employee') : null),
   189→    [formData]
   190→  );
   191→  const supervisorScores = useMemo(
   192→    () => (formData ? calculateAppraisalScores(formData, 'supervisor') : null),
   193→    [formData]
   194→  );
   195→
   196→  // Save draft
   197→  const handleSave = async () => {
   198→    if (!assignmentId || !formData) return;
   199→    setSaving(true);
   200→    try {
   201→      const res = await fetch(`/api/assignments/${assignmentId}/form`, {
   202→        method: 'PUT',
   203→        headers: { 'Content-Type': 'application/json' },
   204→        body: JSON.stringify(formData),
   205→      });
   206→      if (res.ok) {
   207→        toast.success('Draft saved successfully');
   208→      } else {
   209→        const data = await res.json();
   210→        toast.error(data.error || 'Failed to save');
   211→      }
   212→    } catch {
   213→      toast.error('Failed to save');
   214→    } finally {
   215→      setSaving(false);
   216→    }
   217→  };
   218→
   219→  // Determine the workflow action based on current user role
   220→  const getSubmitAction = (): string => {
   221→    if (userRole === 'employee') return 'employee_submit';
   222→    if (userRole === 'supervisor') return 'supervisor_submit';
   223→    if (userRole === 'admin') return 'hr_submit';
   224→    if (userRole === 'management') return 'management_approve';
   225→    return 'employee_submit';
   226→  };
   227→
   228→  // Submit
   229→  const handleSubmit = async () => {
   230→    if (!assignmentId || !formData) return;
   231→
   232→    // Validation
   233→    if (canEditEmployee) {
   234→      const filledGoals = formData.futureGoals.filter((g) => g.description.trim() !== '');
   235→      if (filledGoals.length < 3) {
   236→        toast.error('Minimum 3 goals are required in Section 6');
   237→        return;
   238→      }
   239→    }
   240→
   241→    setSubmitting(true);
   242→    try {
   243→      // Save form data first
   244→      await fetch(`/api/assignments/${assignmentId}/form`, {
   245→        method: 'PUT',
   246→        headers: { 'Content-Type': 'application/json' },
   247→        body: JSON.stringify(formData),
   248→      });
   249→
   250→      const action = getSubmitAction();
   251→      const res = await fetch(`/api/assignments/${assignmentId}/submit`, {
   252→        method: 'POST',
   253→        headers: { 'Content-Type': 'application/json' },
   254→        body: JSON.stringify({ action }),
   255→      });
   256→      if (res.ok) {
   257→        toast.success('Appraisal submitted successfully');
   258→        setCurrentView('appraisal-list');
   259→      } else {
   260→        const data = await res.json();
   261→        toast.error(data.error || 'Failed to submit');
   262→      }
   263→    } catch {
   264→      toast.error('Failed to submit');
   265→    } finally {
   266→      setSubmitting(false);
   267→    }
   268→  };
   269→
   270→  // Return for correction
   271→  const handleReturn = async () => {
   272→    if (!assignmentId || !returnReason.trim()) {
   273→      toast.error('Please provide a reason for returning');
   274→      return;
   275→    }
   276→    setReturning(true);
   277→    try {
   278→      const res = await fetch(`/api/assignments/${assignmentId}/submit`, {
   279→        method: 'POST',
   280→        headers: { 'Content-Type': 'application/json' },
   281→        body: JSON.stringify({ action: 'management_return', returnReason }),
   282→      });
   283→      if (res.ok) {
   284→        toast.success('Appraisal returned for correction');
   285→        setCurrentView('appraisal-list');
   286→      } else {
   287→        const data = await res.json();
   288→        toast.error(data.error || 'Failed to return');
   289→      }
   290→    } catch {
   291→      toast.error('Failed to return');
   292→    } finally {
   293→      setReturning(false);
   294→      setReturnDialog(false);
   295→    }
   296→  };
   297→
   298→  if (loading) {
   299→    return (
   300→      <div className="space-y-4">
   301→        <Skeleton className="h-10 w-64" />
   302→        <Skeleton className="h-64 rounded-xl" />
   303→        <Skeleton className="h-64 rounded-xl" />
   304→        <Skeleton className="h-64 rounded-xl" />
   305→      </div>
   306→    );
   307→  }
   308→
   309→  if (error) {
   310→    return (
   311→      <div className="text-center py-12">
   312→        <AlertTriangle className="h-12 w-12 mx-auto text-amber-500 mb-4" />
   313→        <h3 className="text-lg font-semibold mb-2">Connection Issue</h3>
   314→        <p className="text-muted-foreground mb-4">{error}</p>
   315→        <Button variant="outline" onClick={() => setCurrentView('appraisal-list')}>
   316→          <ArrowLeft className="h-4 w-4 mr-2" />
   317→          Back to List
   318→        </Button>
   319→      </div>
   320→    );
   321→  }
   322→
   323→  if (!assignment || !formData) {
   324→    return <div className="text-center py-12 text-muted-foreground">Assignment not found</div>;
   325→  }
   326→
   327→  const isReadOnly = !canEditEmployee && !canEditSupervisor && !canEditHR && !canEditManagement;
   328→
   329→  // Section header component
   330→  function SectionHeader({ number, title }: { number: number; title: string }) {
   331→    return (
   332→      <div className="eci-gradient-header text-white px-4 py-3 rounded-t-xl">
   333→        <h3 className="font-semibold text-sm">
   334→          Section {number}: {title}
   335→        </h3>
   336→      </div>
   337→    );
   338→  }
   339→
   340→  // Rating select for Section 2
   341→  function GoalRatingSelect({
   342→    value,
   343→    onChange,
   344→    disabled,
   345→  }: {
   346→    value: number | 'NA';
   347→    onChange: (val: number | 'NA') => void;
   348→    disabled: boolean;
   349→  }) {
   350→    return (
   351→      <Select value={String(value)} onValueChange={(v) => onChange(v === 'NA' ? 'NA' : Number(v))} disabled={disabled}>
   352→        <SelectTrigger className="w-28 h-8 text-xs">
   353→          <SelectValue />
   354→        </SelectTrigger>
   355→        <SelectContent>
   356→          {GOAL_RATING_OPTIONS.map((opt) => (
   357→            <SelectItem key={opt.value} value={String(opt.value)}>
   358→              {opt.value} - {opt.label}
   359→            </SelectItem>
   360→          ))}
   361→          <SelectItem value="NA">NA</SelectItem>
   362→        </SelectContent>
   363→      </Select>
   364→    );
   365→  }
   366→
   367→  // Rating select for Section 3
   368→  function CompetencyRatingSelect({
   369→    value,
   370→    onChange,
   371→    disabled,
   372→  }: {
   373→    value: number | 'NA';
   374→    onChange: (val: number | 'NA') => void;
   375→    disabled: boolean;
   376→  }) {
   377→    return (
   378→      <Select value={String(value)} onValueChange={(v) => onChange(v === 'NA' ? 'NA' : Number(v))} disabled={disabled}>
   379→        <SelectTrigger className="w-28 h-8 text-xs">
   380→          <SelectValue />
   381→        </SelectTrigger>
   382→        <SelectContent>
   383→          {COMPETENCY_RATING_OPTIONS.map((opt) => (
   384→            <SelectItem key={String(opt.value)} value={String(opt.value)}>
   385→              {opt.value} - {opt.label}
   386→            </SelectItem>
   387→          ))}
   388→        </SelectContent>
   389→      </Select>
   390→    );
   391→  }
   392→
   393→  return (
   394→    <div className="space-y-6">
   395→      {/* Header */}
   396→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   397→        <div className="flex items-center gap-4">
   398→          <Button variant="ghost" size="sm" onClick={() => setCurrentView('appraisal-list')}>
   399→            <ArrowLeft className="h-4 w-4 mr-2" />
   400→            Back
   401→          </Button>
   402→          <div>
   403→            <h2 className="text-xl font-bold">Performance Appraisal Form</h2>
   404→            <div className="flex items-center gap-3 mt-1">
   405→              <span className="text-sm text-muted-foreground">
   406→                {assignment.employee?.name} · {assignment.cycle?.name}
   407→              </span>
   408→              <Badge className={APPRAISAL_STATUS_COLORS[assignment.status] || ''}>
   409→                {APPRAISAL_STATUS_LABELS[assignment.status] || assignment.status}
   410→              </Badge>
   411→            </div>
   412→          </div>
   413→        </div>
   414→        <div className="flex items-center gap-2 no-print">
   415→          {canSubmit && (
   416→            <Button className="eci-btn-primary" onClick={handleSubmit} disabled={submitting}>
   417→              {submitting ? <Loader2 className="h-4 w-4 mr-2 animate-spin" /> : <Send className="h-4 w-4 mr-2" />}
   418→              Submit
   419→            </Button>
   420→          )}
   421→          {canEditManagement && (
   422→            <Button variant="outline" className="text-amber-600 border-amber-300 hover:bg-amber-50" onClick={() => setReturnDialog(true)}>
   423→              <RotateCcw className="h-4 w-4 mr-2" />
   424→              Return for Correction
   425→            </Button>
   426→          )}
   427→          <Button variant="outline" onClick={handleSave} disabled={saving}>
   428→            {saving ? <Loader2 className="h-4 w-4 mr-2 animate-spin" /> : <Save className="h-4 w-4 mr-2" />}
   429→            Save Draft
   430→          </Button>
   431→        </div>
   432→      </div>
   433→
   434→      {/* Section 1: Basic Information */}
   435→      <Card className="eci-card overflow-hidden">
   436→        <SectionHeader number={1} title="Basic Information" />
   437→        <CardContent className="p-4">
   438→          <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-4 gap-3 text-sm">
   439→            {[
   440→              ['Employee ID', formData.employeeId],
   441→              ['Name', formData.employeeName],
   442→              ['Designation', formData.designation],
   443→              ['Overall Experience', formData.overallExp],
   444→              ['Years with ECI', formData.yearsWithECI],
   445→              ['Current Education', formData.currentEdu],
   446→              ['Required Experience', formData.requiredExp],
   447→              ['Required Education', formData.requiredEdu],
   448→              ['Department', formData.department],
   449→              ['Appraisal Period', formData.appraisalPeriod],
   450→              ['Line Manager Name', formData.lineManagerName],
   451→              ['Line Manager Designation', formData.lineManagerDesignation],
   452→            ].map(([label, value]) => (
   453→              <div key={label as string} className="bg-muted/30 p-2 rounded">
   454→                <p className="text-xs text-muted-foreground">{label as string}</p>
   455→                <p className="font-medium">{value || '-'}</p>
   456→              </div>
   457→            ))}
   458→          </div>
   459→        </CardContent>
   460→      </Card>
   461→
   462→      {/* Section 2: Key Accomplishments and Goals */}
   463→      <Card className="eci-card overflow-hidden">
   464→        <SectionHeader number={2} title="Key Accomplishments and Goals of " />
   465→        <CardContent className="p-4 space-y-4">
   466→          <p className="text-sm text-muted-foreground bg-amber-50 p-3 rounded-lg border border-amber-200">
   467→            {SECTION_DESCRIPTIONS.accomplishments}
   468→          </p>
   469→          <p className="text-xs text-muted-foreground">
   470→            Rating: 0 = No Achievement, 1 = Partial, 2 = Substantial, 3 = Full Achievement
   471→          </p>
   472→
   473→          {/* Achievements */}
   474→          <div>
   475→            <h4 className="font-semibold text-sm mb-2">Achievements</h4>
   476→            <div className="overflow-x-auto">
   477→              <table className="w-full text-sm border-collapse">
   478→                <thead>
   479→                  <tr className="border-b">
   480→                    <th className="text-left p-2 min-w-48">#</th>
   481→                    <th className="text-left p-2 min-w-48">Description</th>
   482→                    <th className="text-center p-2 w-32">Employee Rating</th>
   483→                    <th className="text-center p-2 w-32">Supervisor Rating</th>
   484→                  </tr>
   485→                </thead>
   486→                <tbody>
   487→                  {formData.achievements.map((item, idx) => (
   488→                    <tr key={idx} className="border-b last:border-0">
   489→                      <td className="p-2 text-muted-foreground">{idx + 1}</td>
   490→                      <td className="p-2">
   491→                        <Textarea
   492→                          value={item.description}
   493→                          onChange={(e) => updateAchievement('achievements', idx, 'description', e.target.value)}
   494→                          disabled={!canEditEmployee}
   495→                          placeholder="Describe achievement..."
   496→                          className="min-h-16 text-sm"
   497→                        />
   498→                      </td>
   499→                      <td className="p-2">
   500→                        <GoalRatingSelect
   501→                          value={item.employeeRating}
   502→                          onChange={(v) => updateAchievement('achievements', idx, 'employeeRating', v)}
   503→                          disabled={!canEditEmployee}
   504→                        />
   505→                      </td>
   506→                      <td className="p-2">
   507→                        <GoalRatingSelect
   508→                          value={item.supervisorRating}
   509→                          onChange={(v) => updateAchievement('achievements', idx, 'supervisorRating', v)}
   510→                          disabled={!canEditSupervisor}
   511→                        />
   512→                      </td>
   513→                    </tr>
   514→                  ))}
   515→                  <tr className="bg-muted/20 font-semibold">
   516→                    <td colSpan={2} className="p-2 text-right">Sub-Total (Achievements):</td>
   517→                    <td className="p-2 text-center">
   518→                      {formData.achievements.reduce((s, i) => s + (i.employeeRating === 'NA' ? 0 : Number(i.employeeRating)), 0)} /{' '}
   519→                      {(formData.achievements.length - formData.achievements.filter((i) => i.employeeRating === 'NA').length) * 3}
   520→                    </td>
   521→                    <td className="p-2 text-center">
   522→                      {formData.achievements.reduce((s, i) => s + (i.supervisorRating === 'NA' ? 0 : Number(i.supervisorRating)), 0)} /{' '}
   523→                      {(formData.achievements.length - formData.achievements.filter((i) => i.supervisorRating === 'NA').length) * 3}
   524→                    </td>
   525→                  </tr>
   526→                </tbody>
   527→              </table>
   528→            </div>
   529→          </div>
   530→
   531→          <Separator />
   532→
   533→          {/* Goals */}
   534→          <div>
   535→            <h4 className="font-semibold text-sm mb-2">Goals</h4>
   536→            <div className="overflow-x-auto">
   537→              <table className="w-full text-sm border-collapse">
   538→                <thead>
   539→                  <tr className="border-b">
   540→                    <th className="text-left p-2 min-w-48">#</th>
   541→                    <th className="text-left p-2 min-w-48">Description</th>
   542→                    <th className="text-center p-2 w-32">Employee Rating</th>
   543→                    <th className="text-center p-2 w-32">Supervisor Rating</th>
   544→                  </tr>
   545→                </thead>
   546→                <tbody>
   547→                  {formData.goals.map((item, idx) => (
   548→                    <tr key={idx} className="border-b last:border-0">
   549→                      <td className="p-2 text-muted-foreground">{idx + 1}</td>
   550→                      <td className="p-2">
   551→                        <Textarea
   552→                          value={item.description}
