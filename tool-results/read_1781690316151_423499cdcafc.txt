     1→'use client';
     2→

     4→import { useAppStore } from '@/store/app-store';
     5→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
     6→import { Button } from '@/components/ui/button';
     7→import { Input } from '@/components/ui/input';

     9→import { Label } from '@/components/ui/label';
    10→import { Checkbox } from '@/components/ui/checkbox';

    12→import { Skeleton } from '@/components/ui/skeleton';
    13→import { Separator } from '@/components/ui/separator';
    14→import {
    15→  Select,
    16→  SelectContent,
    17→  SelectItem,
    18→  SelectTrigger,
    19→  SelectValue,
    20→} from '@/components/ui/select';
    21→import {
    22→  AlertDialog,
    23→  AlertDialogAction,
    24→  AlertDialogCancel,
    25→  AlertDialogContent,
    26→  AlertDialogDescription,
    27→  AlertDialogFooter,
    28→  AlertDialogHeader,
    29→  AlertDialogTitle,
    30→} from '@/components/ui/alert-dialog';
    31→import {
    32→  Dialog,
    33→  DialogContent,
    34→  DialogDescription,
    35→  DialogFooter,
    36→  DialogHeader,
    37→  DialogTitle,
    38→} from '@/components/ui/dialog';
    39→import { RadioGroup, RadioGroupItem } from '@/components/ui/radio-group';
    40→import { ArrowLeft, Save, Send, RotateCcw, Loader2 } from 'lucide-react';
    41→import { toast } from 'sonner';
    42→import type {
    43→  AppraisalFormDataFull,
    44→  AssignmentDetail,
    45→  AchievementGoalItem,
    46→  CompetencyItem,
    47→  ExplanationItem,
    48→  FutureGoalItem,
    49→  RemarksData,
    50→  AppraisalStatus,
    51→} from '@/lib/types';
    52→import {
    53→  GOAL_RATING_OPTIONS,
    54→  COMPETENCY_RATING_OPTIONS,
    55→  TECHNICAL_SKILLS,
    56→  LEADERSHIP_SKILLS,
    57→  MANAGERIAL_SKILLS,
    58→  SECTION_DESCRIPTIONS,
    59→  createDefaultFormData,
    60→  calculateAppraisalScores,
    61→} from '@/lib/constants';
    62→import { APPRAISAL_STATUS_LABELS, APPRAISAL_STATUS_COLORS } from '@/lib/constants';
    63→
    64→export default function AppraisalForm() {
    65→  const { viewParams, currentUser, setCurrentView } = useAppStore();
    66→  const assignmentId = viewParams?.id;
    67→
    68→  const [assignment, setAssignment] = useState<AssignmentDetail | null>(null);
    69→  const [formData, setFormData] = useState<AppraisalFormDataFull | null>(null);
    70→  const [loading, setLoading] = useState(true);
    71→  const [saving, setSaving] = useState(false);
    72→  const [submitting, setSubmitting] = useState(false);
    73→  const [returnDialog, setReturnDialog] = useState(false);
    74→  const [returnReason, setReturnReason] = useState('');
    75→  const [returning, setReturning] = useState(false);
    76→
    77→  const userRole = currentUser?.role || 'employee';
    78→  const currentActionBy = assignment?.currentActionBy;
    79→
    80→  // Can the user edit?
    81→  const canEditEmployee = userRole === 'employee' && currentActionBy === 'employee';
    82→  const canEditSupervisor = userRole === 'supervisor' && (currentActionBy === 'supervisor' || currentActionBy === 'employee');
    83→  const canEditHR = (userRole === 'admin' || userRole === 'hr') && currentActionBy === 'hr';
    84→  const canEditManagement = userRole === 'management' && (currentActionBy === 'management');
    85→  const canSubmit = currentActionBy === userRole;
    86→
    87→  // Fetch assignment and form data
    88→  useEffect(() => {
    89→    if (!assignmentId) return;
    90→    async function fetchData() {
    91→      try {
    92→        const [assignRes, formRes] = await Promise.all([
    93→          fetch(`/api/assignments/${assignmentId}`),
    94→          fetch(`/api/assignments/${assignmentId}/form`),
    95→        ]);
    96→        if (assignRes.ok) setAssignment(await assignRes.json());
    97→        if (formRes.ok) {
    98→          const data = await formRes.json();
    99→          setFormData(data.formData || createDefaultFormData());
   100→        } else {
   101→          setFormData(createDefaultFormData());
   102→        }
   103→      } catch {
   104→        toast.error('Failed to load appraisal data');
   105→      } finally {
   106→        setLoading(false);
   107→      }
   108→    }
   109→    fetchData();
   110→  }, [assignmentId]);
   111→
   112→  // Update form helpers
   113→  const updateAchievement = useCallback(
   114→    (section: 'achievements' | 'goals', index: number, field: keyof AchievementGoalItem, value: string | number) => {
   115→      setFormData((prev) => {
   116→        if (!prev) return prev;
   117→        const updated = [...prev[section]];
   118→        updated[index] = { ...updated[index], [field]: value };
   119→        return { ...prev, [section]: updated };
   120→      });
   121→    },
   122→    []
   123→  );
   124→
   125→  const updateSkill = useCallback(
   126→    (section: 'technicalSkills' | 'leadershipSkills' | 'managerialSkills', index: number, field: 'employeeRating' | 'supervisorRating', value: number | 'NA') => {
   127→      setFormData((prev) => {
   128→        if (!prev) return prev;
   129→        const updated = [...prev[section]];
   130→        updated[index] = { ...updated[index], [field]: value };
   131→        return { ...prev, [section]: updated };
   132→      });
   133→    },
   134→    []
   135→  );
   136→
   137→  const updateExplanation = useCallback(
   138→    (index: number, field: keyof ExplanationItem, value: string | boolean | number) => {
   139→      setFormData((prev) => {
   140→        if (!prev) return prev;
   141→        const updated = [...prev.explanations];
   142→        updated[index] = { ...updated[index], [field]: value };
   143→        return { ...prev, explanations: updated };
   144→      });
   145→    },
   146→    []
   147→  );
   148→
   149→  const updateGoal = useCallback(
   150→    (index: number, field: keyof FutureGoalItem, value: string | boolean | [boolean, boolean, boolean, boolean], quarterIdx?: number) => {
   151→      setFormData((prev) => {
   152→        if (!prev) return prev;
   153→        const updated = [...prev.futureGoals];
   154→        if (field === 'quarters' && quarterIdx !== undefined) {
   155→          const newQuarters = [...updated[index].quarters] as [boolean, boolean, boolean, boolean];
   156→          newQuarters[quarterIdx] = value as boolean;
   157→          updated[index] = { ...updated[index], quarters: newQuarters };
   158→        } else {
   159→          updated[index] = { ...updated[index], [field]: value };
   160→        }
   161→        return { ...prev, futureGoals: updated };
   162→      });
   163→    },
   164→    []
   165→  );
   166→
   167→  const updateRemarks = useCallback(
   168→    (section: 'hr' | 'supervisor', field: string, value: boolean | string | null) => {
   169→      setFormData((prev) => {
   170→        if (!prev) return prev;
   171→        const updated = { ...prev.remarks };
   172→        if (section === 'hr') {
   173→          (updated as any)[`hr${field}`] = value;
   174→        } else {
   175→          (updated as any)[`supervisor${field}`] = value;
   176→        }
   177→        return { ...prev, remarks: updated as RemarksData };
   178→      });
   179→    },
   180→    []
   181→  );
   182→
   183→  // Calculate scores
   184→  const employeeScores = useMemo(
   185→    () => (formData ? calculateAppraisalScores(formData, 'employee') : null),
   186→    [formData]
   187→  );
   188→  const supervisorScores = useMemo(
   189→    () => (formData ? calculateAppraisalScores(formData, 'supervisor') : null),
   190→    [formData]
   191→  );
   192→
   193→  // Save draft
   194→  const handleSave = async () => {
   195→    if (!assignmentId || !formData) return;
   196→    setSaving(true);
   197→    try {
   198→      const res = await fetch(`/api/assignments/${assignmentId}/form`, {
   199→        method: 'PUT',
   200→        headers: { 'Content-Type': 'application/json' },
   201→        body: JSON.stringify(formData),
   202→      });
   203→      if (res.ok) {
   204→        toast.success('Draft saved successfully');
   205→      } else {
   206→        const data = await res.json();
   207→        toast.error(data.error || 'Failed to save');
   208→      }
   209→    } catch {
   210→      toast.error('Failed to save');
   211→    } finally {
   212→      setSaving(false);
   213→    }
   214→  };
   215→
   216→  // Determine the workflow action based on current user role
   217→  const getSubmitAction = (): string => {
   218→    if (userRole === 'employee') return 'employee_submit';
   219→    if (userRole === 'supervisor') return 'supervisor_submit';
   220→    if (userRole === 'admin') return 'hr_submit';
   221→    if (userRole === 'management') return 'management_approve';
   222→    return 'employee_submit';
   223→  };
   224→
   225→  // Submit
   226→  const handleSubmit = async () => {
   227→    if (!assignmentId || !formData) return;
   228→
   229→    // Validation
   230→    if (canEditEmployee) {
   231→      const filledGoals = formData.futureGoals.filter((g) => g.description.trim() !== '');
   232→      if (filledGoals.length < 3) {
   233→        toast.error('Minimum 3 goals are required in Section 6');
   234→        return;
   235→      }
   236→    }
   237→
   238→    setSubmitting(true);
   239→    try {
   240→      // Save form data first
   241→      await fetch(`/api/assignments/${assignmentId}/form`, {
   242→        method: 'PUT',
   243→        headers: { 'Content-Type': 'application/json' },
   244→        body: JSON.stringify(formData),
   245→      });
   246→
   247→      const action = getSubmitAction();
   248→      const res = await fetch(`/api/assignments/${assignmentId}/submit`, {
   249→        method: 'POST',
   250→        headers: { 'Content-Type': 'application/json' },
   251→        body: JSON.stringify({ action }),
   252→      });
   253→      if (res.ok) {
   254→        toast.success('Appraisal submitted successfully');
   255→        setCurrentView('appraisal-list');
   256→      } else {
   257→        const data = await res.json();
   258→        toast.error(data.error || 'Failed to submit');
   259→      }
   260→    } catch {
   261→      toast.error('Failed to submit');
   262→    } finally {
   263→      setSubmitting(false);
   264→    }
   265→  };
   266→
   267→  // Return for correction
   268→  const handleReturn = async () => {
   269→    if (!assignmentId || !returnReason.trim()) {
   270→      toast.error('Please provide a reason for returning');
   271→      return;
   272→    }
   273→    setReturning(true);
   274→    try {
   275→      const res = await fetch(`/api/assignments/${assignmentId}/submit`, {
   276→        method: 'POST',
   277→        headers: { 'Content-Type': 'application/json' },
   278→        body: JSON.stringify({ action: 'management_return', returnReason }),
   279→      });
   280→      if (res.ok) {
   281→        toast.success('Appraisal returned for correction');
   282→        setCurrentView('appraisal-list');
   283→      } else {
   284→        const data = await res.json();
   285→        toast.error(data.error || 'Failed to return');
   286→      }
   287→    } catch {
   288→      toast.error('Failed to return');
   289→    } finally {
   290→      setReturning(false);
   291→      setReturnDialog(false);
   292→    }
   293→  };
   294→
   295→  if (loading) {
   296→    return (
   297→      <div className="space-y-4">
   298→        <Skeleton className="h-10 w-64" />
   299→        <Skeleton className="h-64 rounded-xl" />
   300→        <Skeleton className="h-64 rounded-xl" />
   301→        <Skeleton className="h-64 rounded-xl" />
   302→      </div>
   303→    );
   304→  }
   305→
   306→  if (!assignment || !formData) {
   307→    return <div className="text-center py-12 text-muted-foreground">Assignment not found</div>;
   308→  }
   309→
   310→  const isReadOnly = !canEditEmployee && !canEditSupervisor && !canEditHR && !canEditManagement;
   311→
   312→  // Section header component
   313→  function SectionHeader({ number, title }: { number: number; title: string }) {
   314→    return (
   315→      <div className="eci-gradient-header text-white px-4 py-3 rounded-t-xl">
   316→        <h3 className="font-semibold text-sm">
   317→          Section {number}: {title}
   318→        </h3>
   319→      </div>
   320→    );
   321→  }
   322→
   323→  // Rating select for Section 2
   324→  function GoalRatingSelect({
   325→    value,
   326→    onChange,
   327→    disabled,
   328→  }: {
   329→    value: number | 'NA';
   330→    onChange: (val: number | 'NA') => void;
   331→    disabled: boolean;
   332→  }) {
   333→    return (
   334→      <Select value={String(value)} onValueChange={(v) => onChange(v === 'NA' ? 'NA' : Number(v))} disabled={disabled}>
   335→        <SelectTrigger className="w-28 h-8 text-xs">
   336→          <SelectValue />
   337→        </SelectTrigger>
   338→        <SelectContent>
   339→          {GOAL_RATING_OPTIONS.map((opt) => (
   340→            <SelectItem key={opt.value} value={String(opt.value)}>
   341→              {opt.value} - {opt.label}
   342→            </SelectItem>
   343→          ))}
   344→          <SelectItem value="NA">NA</SelectItem>
   345→        </SelectContent>
   346→      </Select>
   347→    );
   348→  }
   349→
   350→  // Rating select for Section 3
   351→  function CompetencyRatingSelect({
   352→    value,
   353→    onChange,
   354→    disabled,
   355→  }: {
   356→    value: number | 'NA';
   357→    onChange: (val: number | 'NA') => void;
   358→    disabled: boolean;
   359→  }) {
   360→    return (
   361→      <Select value={String(value)} onValueChange={(v) => onChange(v === 'NA' ? 'NA' : Number(v))} disabled={disabled}>
   362→        <SelectTrigger className="w-28 h-8 text-xs">
   363→          <SelectValue />
   364→        </SelectTrigger>
   365→        <SelectContent>
   366→          {COMPETENCY_RATING_OPTIONS.map((opt) => (
   367→            <SelectItem key={String(opt.value)} value={String(opt.value)}>
   368→              {opt.value} - {opt.label}
   369→            </SelectItem>
   370→          ))}
   371→        </SelectContent>
   372→      </Select>
   373→    );
   374→  }
   375→
   376→  return (
   377→    <div className="space-y-6">
   378→      {/* Header */}
   379→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   380→        <div className="flex items-center gap-4">
   381→          <Button variant="ghost" size="sm" onClick={() => setCurrentView('appraisal-list')}>
   382→            <ArrowLeft className="h-4 w-4 mr-2" />
   383→            Back
   384→          </Button>
   385→          <div>
   386→            <h2 className="text-xl font-bold">Performance Appraisal Form</h2>
   387→            <div className="flex items-center gap-3 mt-1">
   388→              <span className="text-sm text-muted-foreground">
   389→                {assignment.employee?.name} · {assignment.cycle?.name}
   390→              </span>
   391→              <Badge className={APPRAISAL_STATUS_COLORS[assignment.status] || ''}>
   392→                {APPRAISAL_STATUS_LABELS[assignment.status] || assignment.status}
   393→              </Badge>
   394→            </div>
   395→          </div>
   396→        </div>
   397→        <div className="flex items-center gap-2 no-print">
   398→          {canSubmit && (
   399→            <Button className="eci-btn-primary" onClick={handleSubmit} disabled={submitting}>
   400→              {submitting ? <Loader2 className="h-4 w-4 mr-2 animate-spin" /> : <Send className="h-4 w-4 mr-2" />}
   401→              Submit
   402→            </Button>
   403→          )}
   404→          {canEditManagement && (
   405→            <Button variant="outline" className="text-amber-600 border-amber-300 hover:bg-amber-50" onClick={() => setReturnDialog(true)}>
   406→              <RotateCcw className="h-4 w-4 mr-2" />
   407→              Return for Correction
   408→            </Button>
   409→          )}
   410→          <Button variant="outline" onClick={handleSave} disabled={saving}>
   411→            {saving ? <Loader2 className="h-4 w-4 mr-2 animate-spin" /> : <Save className="h-4 w-4 mr-2" />}
   412→            Save Draft
   413→          </Button>
   414→        </div>
   415→      </div>
   416→
   417→      {/* Section 1: Basic Information */}
   418→      <Card className="eci-card overflow-hidden">
   419→        <SectionHeader number={1} title="Basic Information" />
   420→        <CardContent className="p-4">
   421→          <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-4 gap-3 text-sm">
   422→            {[
   423→              ['Employee ID', formData.employeeId],
   424→              ['Name', formData.employeeName],
   425→              ['Designation', formData.designation],
   426→              ['Overall Experience', formData.overallExp],
   427→              ['Years with ECI', formData.yearsWithECI],
   428→              ['Current Education', formData.currentEdu],
   429→              ['Required Experience', formData.requiredExp],
   430→              ['Required Education', formData.requiredEdu],
   431→              ['Department', formData.department],
   432→              ['Appraisal Period', formData.appraisalPeriod],
   433→              ['Line Manager Name', formData.lineManagerName],
   434→              ['Line Manager Designation', formData.lineManagerDesignation],
   435→            ].map(([label, value]) => (
   436→              <div key={label as string} className="bg-muted/30 p-2 rounded">
   437→                <p className="text-xs text-muted-foreground">{label as string}</p>
   438→                <p className="font-medium">{value || '-'}</p>
   439→              </div>
   440→            ))}
   441→          </div>
   442→        </CardContent>
   443→      </Card>
   444→
   445→      {/* Section 2: Key Accomplishments and Goals */}
   446→      <Card className="eci-card overflow-hidden">
   447→        <SectionHeader number={2} title="Key Accomplishments and Goals of " />
   448→        <CardContent className="p-4 space-y-4">
   449→          <p className="text-sm text-muted-foreground bg-amber-50 p-3 rounded-lg border border-amber-200">
   450→            {SECTION_DESCRIPTIONS.accomplishments}
   451→          </p>
   452→          <p className="text-xs text-muted-foreground">
   453→            Rating: 0 = No Achievement, 1 = Partial, 2 = Substantial, 3 = Full Achievement
   454→          </p>
   455→
   456→          {/* Achievements */}
   457→          <div>
   458→            <h4 className="font-semibold text-sm mb-2">Achievements</h4>
   459→            <div className="overflow-x-auto">
   460→              <table className="w-full text-sm border-collapse">
   461→                <thead>
   462→                  <tr className="border-b">
   463→                    <th className="text-left p-2 min-w-48">#</th>
   464→                    <th className="text-left p-2 min-w-48">Description</th>
   465→                    <th className="text-center p-2 w-32">Employee Rating</th>
   466→                    <th className="text-center p-2 w-32">Supervisor Rating</th>
   467→                  </tr>
   468→                </thead>
   469→                <tbody>
   470→                  {formData.achievements.map((item, idx) => (
   471→                    <tr key={idx} className="border-b last:border-0">
   472→                      <td className="p-2 text-muted-foreground">{idx + 1}</td>
   473→                      <td className="p-2">
   474→                        <Textarea
   475→                          value={item.description}
   476→                          onChange={(e) => updateAchievement('achievements', idx, 'description', e.target.value)}
   477→                          disabled={!canEditEmployee}
   478→                          placeholder="Describe achievement..."
   479→                          className="min-h-16 text-sm"
   480→                        />
   481→                      </td>
   482→                      <td className="p-2">
   483→                        <GoalRatingSelect
   484→                          value={item.employeeRating}
   485→                          onChange={(v) => updateAchievement('achievements', idx, 'employeeRating', v)}
   486→                          disabled={!canEditEmployee}
   487→                        />
   488→                      </td>
   489→                      <td className="p-2">
   490→                        <GoalRatingSelect
   491→                          value={item.supervisorRating}
   492→                          onChange={(v) => updateAchievement('achievements', idx, 'supervisorRating', v)}
   493→                          disabled={!canEditSupervisor}
   494→                        />
   495→                      </td>
   496→                    </tr>
   497→                  ))}
   498→                  <tr className="bg-muted/20 font-semibold">
   499→                    <td colSpan={2} className="p-2 text-right">Sub-Total (Achievements):</td>
   500→                    <td className="p-2 text-center">
   501→                      {formData.achievements.reduce((s, i) => s + (i.employeeRating === 'NA' ? 0 : Number(i.employeeRating)), 0)} /{' '}
   502→                      {(formData.achievements.length - formData.achievements.filter((i) => i.employeeRating === 'NA').length) * 3}
   503→                    </td>
   504→                    <td className="p-2 text-center">
   505→                      {formData.achievements.reduce((s, i) => s + (i.supervisorRating === 'NA' ? 0 : Number(i.supervisorRating)), 0)} /{' '}
   506→                      {(formData.achievements.length - formData.achievements.filter((i) => i.supervisorRating === 'NA').length) * 3}
   507→                    </td>
   508→                  </tr>
   509→                </tbody>
   510→              </table>
   511→            </div>
   512→          </div>
   513→
   514→          <Separator />
   515→
   516→          {/* Goals */}
   517→          <div>
   518→            <h4 className="font-semibold text-sm mb-2">Goals</h4>
   519→            <div className="overflow-x-auto">
   520→              <table className="w-full text-sm border-collapse">
   521→                <thead>
   522→                  <tr className="border-b">
   523→                    <th className="text-left p-2 min-w-48">#</th>
   524→                    <th className="text-left p-2 min-w-48">Description</th>
   525→                    <th className="text-center p-2 w-32">Employee Rating</th>
   526→                    <th className="text-center p-2 w-32">Supervisor Rating</th>
   527→                  </tr>
   528→                </thead>
   529→                <tbody>
   530→                  {formData.goals.map((item, idx) => (
   531→                    <tr key={idx} className="border-b last:border-0">
   532→                      <td className="p-2 text-muted-foreground">{idx + 1}</td>
   533→                      <td className="p-2">
   534→                        <Textarea
   535→                          value={item.description}
   536→                          onChange={(e) => updateAchievement('goals', idx, 'description', e.target.value)}
   537→                          disabled={!canEditEmployee}
   538→                          placeholder="Describe goal..."
   539→                          className="min-h-16 text-sm"
   540→                        />
   541→                      </td>
   542→                      <td className="p-2">
   543→                        <GoalRatingSelect
   544→                          value={item.employeeRating}
   545→                          onChange={(v) => updateAchievement('goals', idx, 'employeeRating', v)}
   546→                          disabled={!canEditEmployee}
   547→                        />
   548→                      </td>
   549→                      <td className="p-2">
   550→                        <GoalRatingSelect
   551→                          value={item.supervisorRating}
   552→                          onChange={(v) => updateAchievement('goals', idx, 'supervisorRating', v)}
   553→                          disabled={!canEditSupervisor}
   554→                        />
   555→                      </td>
   556→                    </tr>
