    21→import {

    30→} from '@/components/ui/alert-dialog';
    31→import {

    40→import { ArrowLeft, Save, Send, RotateCcw, Loader2 } from 'lucide-react';
    41→import { toast } from 'sonner';
    42→import type {
    43→  AppraisalFormDataFull,
    44→  AssignmentDetail,
    45→  AchievementGoalItem,
    46→  CompetencyItem,
    47→  ExplanationItem,
    48→  FutureGoalItem,
    49→  RemarksData,
    50→  AppraisalStatus,
    51→} from '@/lib/types';
    52→import {
    53→  GOAL_RATING_OPTIONS,
    54→  COMPETENCY_RATING_OPTIONS,
    55→  TECHNICAL_SKILLS,
    56→  LEADERSHIP_SKILLS,
    57→  MANAGERIAL_SKILLS,
    58→  SECTION_DESCRIPTIONS,
    59→  createDefaultFormData,
    60→  calculateAppraisalScores,
    61→} from '@/lib/constants';
    62→import { APPRAISAL_STATUS_LABELS, APPRAISAL_STATUS_COLORS } from '@/lib/constants';
    63→
    64→export default function AppraisalForm() {
    65→  const { viewParams, currentUser, setCurrentView } = useAppStore();
    66→  const assignmentId = viewParams?.id;
    67→
    68→  const [assignment, setAssignment] = useState<AssignmentDetail | null>(null);
    69→  const [formData, setFormData] = useState<AppraisalFormDataFull | null>(null);
    70→  const [loading, setLoading] = useState(true);
    71→  const [saving, setSaving] = useState(false);
    72→  const [submitting, setSubmitting] = useState(false);
    73→  const [returnDialog, setReturnDialog] = useState(false);
    74→  const [returnReason, setReturnReason] = useState('');
    75→  const [returning, setReturning] = useState(false);
    76→
    77→  const userRole = currentUser?.role || 'employee';
    78→  const currentActionBy = assignment?.currentActionBy;
    79→
    80→  // Can the user edit?
    81→  const canEditEmployee = userRole === 'employee' && currentActionBy === 'employee';
    82→  const canEditSupervisor = userRole === 'supervisor' && (currentActionBy === 'supervisor' || currentActionBy === 'employee');
    83→  const canEditHR = (userRole === 'admin' || userRole === 'hr') && currentActionBy === 'hr';
    84→  const canEditManagement = userRole === 'management' && (currentActionBy === 'management');
    85→  const canSubmit = currentActionBy === userRole;
    86→
    87→  // Fetch assignment and form data
    88→  useEffect(() => {
    89→    if (!assignmentId) return;
    90→    async function fetchData() {
    91→      try {
    92→        const [assignRes, formRes] = await Promise.all([
    93→          fetch(`/api/assignments/${assignmentId}`),
    94→          fetch(`/api/assignments/${assignmentId}/form`),
    95→        ]);
    96→        if (assignRes.ok) setAssignment(await assignRes.json());
    97→        if (formRes.ok) {
    98→          const data = await formRes.json();
    99→          setFormData(data.formData || createDefaultFormData());
   100→        } else {
   101→          setFormData(createDefaultFormData());
   102→        }
   103→      } catch {
   104→        toast.error('Failed to load appraisal data');
   105→      } finally {
   106→        setLoading(false);
   107→      }
   108→    }
   109→    fetchData();
   110→  }, [assignmentId]);
   111→
   112→  // Update form helpers
   113→  const updateAchievement = useCallback(
   114→    (section: 'achievements' | 'goals', index: number, field: keyof AchievementGoalItem, value: string | number) => {
   115→      setFormData((prev) => {
   116→        if (!prev) return prev;
   117→        const updated = [...prev[section]];
   118→        updated[index] = { ...updated[index], [field]: value };
   119→        return { ...prev, [section]: updated };
   120→      });
   121→    },
   122→    []
   123→  );
   124→
   125→  const updateSkill = useCallback(
   126→    (section: 'technicalSkills' | 'leadershipSkills' | 'managerialSkills', index: number, field: 'employeeRating' | 'supervisorRating', value: number | 'NA') => {
   127→      setFormData((prev) => {
   128→        if (!prev) return prev;
   129→        const updated = [...prev[section]];
   130→        updated[index] = { ...updated[index], [field]: value };
   131→        return { ...prev, [section]: updated };
   132→      });
   133→    },
   134→    []
   135→  );
   136→
   137→  const updateExplanation = useCallback(
   138→    (index: number, field: keyof ExplanationItem, value: string | boolean | number) => {
   139→      setFormData((prev) => {
   140→        if (!prev) return prev;
   141→        const updated = [...prev.explanations];
   142→        updated[index] = { ...updated[index], [field]: value };
   143→        return { ...prev, explanations: updated };
   144→      });
   145→    },
   146→    []
   147→  );
   148→
   149→  const updateGoal = useCallback(
   150→    (index: number, field: keyof FutureGoalItem, value: string | boolean | [boolean, boolean, boolean, boolean], quarterIdx?: number) => {
   151→      setFormData((prev) => {
   152→        if (!prev) return prev;
   153→        const updated = [...prev.futureGoals];
   154→        if (field === 'quarters' && quarterIdx !== undefined) {
   155→          const newQuarters = [...updated[index].quarters] as [boolean, boolean, boolean, boolean];
   156→          newQuarters[quarterIdx] = value as boolean;
   157→          updated[index] = { ...updated[index], quarters: newQuarters };
   158→        } else {
   159→          updated[index] = { ...updated[index], [field]: value };
   160→        }
   161→        return { ...prev, futureGoals: updated };
   162→      });
   163→    },
   164→    []
   165→  );
   166→
   167→  const updateRemarks = useCallback(
   168→    (section: 'hr' | 'supervisor', field: string, value: boolean | string | null) => {
   169→      setFormData((prev) => {
   170→        if (!prev) return prev;
   171→        const updated = { ...prev.remarks };
   172→        if (section === 'hr') {
   173→          (updated as any)[`hr${field}`] = value;
   174→        } else {
   175→          (updated as any)[`supervisor${field}`] = value;
   176→        }
   177→        return { ...prev, remarks: updated as RemarksData };
   178→      });
   179→    },
   180→    []
   181→  );
   182→
   183→  // Calculate scores
   184→  const employeeScores = useMemo(
   185→    () => (formData ? calculateAppraisalScores(formData, 'employee') : null),
   186→    [formData]
   187→  );
   188→  const supervisorScores = useMemo(
   189→    () => (formData ? calculateAppraisalScores(formData, 'supervisor') : null),
   190→    [formData]
   191→  );
   192→
   193→  // Save draft
   194→  const handleSave = async () => {
   195→    if (!assignmentId || !formData) return;
   196→    setSaving(true);
   197→    try {
   198→      const res = await fetch(`/api/assignments/${assignmentId}/form`, {
   199→        method: 'PUT',
   200→        headers: { 'Content-Type': 'application/json' },
   201→        body: JSON.stringify(formData),
   202→      });
   203→      if (res.ok) {
   204→        toast.success('Draft saved successfully');
   205→      } else {
   206→        const data = await res.json();
   207→        toast.error(data.error || 'Failed to save');
   208→      }
   209→    } catch {
   210→      toast.error('Failed to save');
   211→    } finally {
   212→      setSaving(false);
   213→    }
   214→  };
   215→
   216→  // Submit
   217→  const handleSubmit = async () => {
   218→    if (!assignmentId || !formData) return;
   219→
   220→    // Validation
   221→    if (canEditEmployee) {
   222→      const filledGoals = formData.futureGoals.filter((g) => g.description.trim() !== '');
   223→      if (filledGoals.length < 3) {
   224→        toast.error('Minimum 3 goals are required in Section 6');
   225→        return;
   226→      }
   227→    }
   228→
   229→    setSubmitting(true);
   230→    try {
   231→      // Save form data first
   232→      await fetch(`/api/assignments/${assignmentId}/form`, {
   233→        method: 'PUT',
   234→        headers: { 'Content-Type': 'application/json' },
   235→        body: JSON.stringify(formData),
   236→      });
   237→
   238→      const res = await fetch(`/api/assignments/${assignmentId}/submit`, { method: 'POST' });
   239→      if (res.ok) {
   240→        toast.success('Appraisal submitted successfully');
   241→        setCurrentView('appraisal-list');
   242→      } else {
   243→        const data = await res.json();
   244→        toast.error(data.error || 'Failed to submit');
   245→      }
   246→    } catch {
   247→      toast.error('Failed to submit');
   248→    } finally {
   249→      setSubmitting(false);
   250→    }
   251→  };
   252→
   253→  // Return for correction
   254→  const handleReturn = async () => {
   255→    if (!assignmentId || !returnReason.trim()) {
   256→      toast.error('Please provide a reason for returning');
   257→      return;
   258→    }
   259→    setReturning(true);
   260→    try {
   261→      const res = await fetch(`/api/assignments/${assignmentId}/return`, {
   262→        method: 'POST',
   263→        headers: { 'Content-Type': 'application/json' },
   264→        body: JSON.stringify({ reason: returnReason }),
   265→      });
   266→      if (res.ok) {
   267→        toast.success('Appraisal returned for correction');
   268→        setCurrentView('appraisal-list');
   269→      } else {
   270→        const data = await res.json();
   271→        toast.error(data.error || 'Failed to return');
   272→      }
   273→    } catch {
   274→      toast.error('Failed to return');
   275→    } finally {
   276→      setReturning(false);
   277→      setReturnDialog(false);
   278→    }
   279→  };
   280→
   281→  if (loading) {
   282→    return (
   283→      <div className="space-y-4">
   284→        <Skeleton className="h-10 w-64" />
   285→        <Skeleton className="h-64 rounded-xl" />
   286→        <Skeleton className="h-64 rounded-xl" />
   287→        <Skeleton className="h-64 rounded-xl" />
   288→      </div>
   289→    );
   290→  }
   291→
   292→  if (!assignment || !formData) {
   293→    return <div className="text-center py-12 text-muted-foreground">Assignment not found</div>;
   294→  }
   295→
   296→  const isReadOnly = !canEditEmployee && !canEditSupervisor && !canEditHR && !canEditManagement;
   297→
   298→  // Section header component
   299→  function SectionHeader({ number, title }: { number: number; title: string }) {
   300→    return (
   301→      <div className="eci-gradient-header text-white px-4 py-3 rounded-t-xl">
   302→        <h3 className="font-semibold text-sm">
   303→          Section {number}: {title}
   304→        </h3>
   305→      </div>
   306→    );
   307→  }
   308→
   309→  // Rating select for Section 2
   310→  function GoalRatingSelect({
   311→    value,
   312→    onChange,
   313→    disabled,
   314→  }: {
   315→    value: number | 'NA';
   316→    onChange: (val: number | 'NA') => void;
   317→    disabled: boolean;
   318→  }) {
   319→    return (
   320→      <Select value={String(value)} onValueChange={(v) => onChange(v === 'NA' ? 'NA' : Number(v))} disabled={disabled}>
   321→        <SelectTrigger className="w-28 h-8 text-xs">
   322→          <SelectValue />
   323→        </SelectTrigger>
   324→        <SelectContent>
   325→          {GOAL_RATING_OPTIONS.map((opt) => (
   326→            <SelectItem key={opt.value} value={String(opt.value)}>
   327→              {opt.value} - {opt.label}
   328→            </SelectItem>
   329→          ))}
   330→          <SelectItem value="NA">NA</SelectItem>
   331→        </SelectContent>
   332→      </Select>
   333→    );
   334→  }
   335→
   336→  // Rating select for Section 3
   337→  function CompetencyRatingSelect({
   338→    value,
   339→    onChange,
   340→    disabled,
   341→  }: {
   342→    value: number | 'NA';
   343→    onChange: (val: number | 'NA') => void;
   344→    disabled: boolean;
   345→  }) {
   346→    return (
   347→      <Select value={String(value)} onValueChange={(v) => onChange(v === 'NA' ? 'NA' : Number(v))} disabled={disabled}>
   348→        <SelectTrigger className="w-28 h-8 text-xs">
   349→          <SelectValue />
   350→        </SelectTrigger>
   351→        <SelectContent>
   352→          {COMPETENCY_RATING_OPTIONS.map((opt) => (
   353→            <SelectItem key={String(opt.value)} value={String(opt.value)}>
   354→              {opt.value} - {opt.label}
   355→            </SelectItem>
   356→          ))}
   357→        </SelectContent>
   358→      </Select>
   359→    );
   360→  }
   361→
   362→  return (
   363→    <div className="space-y-6">
   364→      {/* Header */}
   365→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   366→        <div className="flex items-center gap-4">
   367→          <Button variant="ghost" size="sm" onClick={() => setCurrentView('appraisal-list')}>
   368→            <ArrowLeft className="h-4 w-4 mr-2" />
   369→            Back
   370→          </Button>
   371→          <div>
   372→            <h2 className="text-xl font-bold">Performance Appraisal Form</h2>
   373→            <div className="flex items-center gap-3 mt-1">
   374→              <span className="text-sm text-muted-foreground">
   375→                {assignment.employee?.name} · {assignment.cycle?.name}
   376→              </span>
   377→              <Badge className={APPRAISAL_STATUS_COLORS[assignment.status] || ''}>
   378→                {APPRAISAL_STATUS_LABELS[assignment.status] || assignment.status}
   379→              </Badge>
   380→            </div>
   381→          </div>
   382→        </div>
   383→        <div className="flex items-center gap-2 no-print">
   384→          {canSubmit && (
   385→            <Button className="eci-btn-primary" onClick={handleSubmit} disabled={submitting}>
   386→              {submitting ? <Loader2 className="h-4 w-4 mr-2 animate-spin" /> : <Send className="h-4 w-4 mr-2" />}
   387→              Submit
   388→            </Button>
   389→          )}
   390→          {canEditManagement && (
   391→            <Button variant="outline" className="text-amber-600 border-amber-300 hover:bg-amber-50" onClick={() => setReturnDialog(true)}>
   392→              <RotateCcw className="h-4 w-4 mr-2" />
   393→              Return for Correction
   394→            </Button>
   395→          )}
   396→          <Button variant="outline" onClick={handleSave} disabled={saving}>
   397→            {saving ? <Loader2 className="h-4 w-4 mr-2 animate-spin" /> : <Save className="h-4 w-4 mr-2" />}
   398→            Save Draft
   399→          </Button>
   400→        </div>
   401→      </div>
   402→
   403→      {/* Section 1: Basic Information */}
   404→      <Card className="eci-card overflow-hidden">
   405→        <SectionHeader number={1} title="Basic Information" />
   406→        <CardContent className="p-4">
   407→          <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-4 gap-3 text-sm">
   408→            {[
   409→              ['Employee ID', formData.employeeId],
   410→              ['Name', formData.employeeName],
   411→              ['Designation', formData.designation],
   412→              ['Overall Experience', formData.overallExp],
   413→              ['Years with ECI', formData.yearsWithECI],
   414→              ['Current Education', formData.currentEdu],
   415→              ['Required Experience', formData.requiredExp],
   416→              ['Required Education', formData.requiredEdu],
   417→              ['Department', formData.department],
   418→              ['Appraisal Period', formData.appraisalPeriod],
   419→              ['Line Manager Name', formData.lineManagerName],
   420→              ['Line Manager Designation', formData.lineManagerDesignation],
   421→            ].map(([label, value]) => (
   422→              <div key={label as string} className="bg-muted/30 p-2 rounded">
   423→                <p className="text-xs text-muted-foreground">{label as string}</p>
   424→                <p className="font-medium">{value || '-'}</p>
   425→              </div>
   426→            ))}
   427→          </div>
   428→        </CardContent>
   429→      </Card>
   430→
   431→      {/* Section 2: Key Accomplishments and Goals */}
   432→      <Card className="eci-card overflow-hidden">
   433→        <SectionHeader number={2} title="Key Accomplishments and Goals of " />
   434→        <CardContent className="p-4 space-y-4">
   435→          <p className="text-sm text-muted-foreground bg-amber-50 p-3 rounded-lg border border-amber-200">
   436→            {SECTION_DESCRIPTIONS.accomplishments}
   437→          </p>
   438→          <p className="text-xs text-muted-foreground">
   439→            Rating: 0 = No Achievement, 1 = Partial, 2 = Substantial, 3 = Full Achievement
   440→          </p>
   441→
   442→          {/* Achievements */}
   443→          <div>
   444→            <h4 className="font-semibold text-sm mb-2">Achievements</h4>
   445→            <div className="overflow-x-auto">
   446→              <table className="w-full text-sm border-collapse">
   447→                <thead>
   448→                  <tr className="border-b">
   449→                    <th className="text-left p-2 min-w-48">#</th>
   450→                    <th className="text-left p-2 min-w-48">Description</th>
   451→                    <th className="text-center p-2 w-32">Employee Rating</th>
   452→                    <th className="text-center p-2 w-32">Supervisor Rating</th>
   453→                  </tr>
   454→                </thead>
   455→                <tbody>
   456→                  {formData.achievements.map((item, idx) => (
   457→                    <tr key={idx} className="border-b last:border-0">
   458→                      <td className="p-2 text-muted-foreground">{idx + 1}</td>
   459→                      <td className="p-2">
   460→                        <Textarea
   461→                          value={item.description}
   462→                          onChange={(e) => updateAchievement('achievements', idx, 'description', e.target.value)}
   463→                          disabled={!canEditEmployee}
   464→                          placeholder="Describe achievement..."
   465→                          className="min-h-16 text-sm"
   466→                        />
   467→                      </td>
   468→                      <td className="p-2">
   469→                        <GoalRatingSelect
   470→                          value={item.employeeRating}
   471→                          onChange={(v) => updateAchievement('achievements', idx, 'employeeRating', v)}
   472→                          disabled={!canEditEmployee}
   473→                        />
   474→                      </td>
   475→                      <td className="p-2">
   476→                        <GoalRatingSelect
   477→                          value={item.supervisorRating}
   478→                          onChange={(v) => updateAchievement('achievements', idx, 'supervisorRating', v)}
   479→                          disabled={!canEditSupervisor}
   480→                        />
   481→                      </td>
   482→                    </tr>
   483→                  ))}
   484→                  <tr className="bg-muted/20 font-semibold">
   485→                    <td colSpan={2} className="p-2 text-right">Sub-Total (Achievements):</td>
   486→                    <td className="p-2 text-center">
   487→                      {formData.achievements.reduce((s, i) => s + (i.employeeRating === 'NA' ? 0 : Number(i.employeeRating)), 0)} /{' '}
   488→                      {(formData.achievements.length - formData.achievements.filter((i) => i.employeeRating === 'NA').length) * 3}
   489→                    </td>
   490→                    <td className="p-2 text-center">
   491→                      {formData.achievements.reduce((s, i) => s + (i.supervisorRating === 'NA' ? 0 : Number(i.supervisorRating)), 0)} /{' '}
   492→                      {(formData.achievements.length - formData.achievements.filter((i) => i.supervisorRating === 'NA').length) * 3}
   493→                    </td>
   494→                  </tr>
   495→                </tbody>
   496→              </table>
   497→            </div>
   498→          </div>
   499→
   500→          <Separator />
   501→
   502→          {/* Goals */}
   503→          <div>
   504→            <h4 className="font-semibold text-sm mb-2">Goals</h4>
   505→            <div className="overflow-x-auto">
   506→              <table className="w-full text-sm border-collapse">
   507→                <thead>
   508→                  <tr className="border-b">
   509→                    <th className="text-left p-2 min-w-48">#</th>
   510→                    <th className="text-left p-2 min-w-48">Description</th>
   511→                    <th className="text-center p-2 w-32">Employee Rating</th>
   512→                    <th className="text-center p-2 w-32">Supervisor Rating</th>
   513→                  </tr>
   514→                </thead>
   515→                <tbody>
   516→                  {formData.goals.map((item, idx) => (
   517→                    <tr key={idx} className="border-b last:border-0">
   518→                      <td className="p-2 text-muted-foreground">{idx + 1}</td>
   519→                      <td className="p-2">
   520→                        <Textarea
   521→                          value={item.description}
   522→                          onChange={(e) => updateAchievement('goals', idx, 'description', e.target.value)}
   523→                          disabled={!canEditEmployee}
   524→                          placeholder="Describe goal..."
   525→                          className="min-h-16 text-sm"
   526→                        />
   527→                      </td>
   528→                      <td className="p-2">
   529→                        <GoalRatingSelect
   530→                          value={item.employeeRating}
   531→                          onChange={(v) => updateAchievement('goals', idx, 'employeeRating', v)}
   532→                          disabled={!canEditEmployee}
   533→                        />
   534→                      </td>
   535→                      <td className="p-2">
   536→                        <GoalRatingSelect
   537→                          value={item.supervisorRating}
   538→                          onChange={(v) => updateAchievement('goals', idx, 'supervisorRating', v)}
   539→                          disabled={!canEditSupervisor}
   540→                        />
   541→                      </td>
   542→                    </tr>
   543→                  ))}
   544→                  <tr className="bg-muted/20 font-semibold">
   545→                    <td colSpan={2} className="p-2 text-right">Sub-Total (Goals):</td>
   546→                    <td className="p-2 text-center">
   547→                      {formData.goals.reduce((s, i) => s + (i.employeeRating === 'NA' ? 0 : Number(i.employeeRating)), 0)} /{' '}
   548→                      {(formData.goals.length - formData.goals.filter((i) => i.employeeRating === 'NA').length) * 3}
